# ONNX Runtime GPU path (CV_BACKEND=ort). `onnxruntime-gpu` is a distinct wheel from
# `onnxruntime`; installing both shadows one another, so only this one is present.
#
# Version pin is load-bearing. The runtime stage is `nvidia/cuda:12.4.1-cudnn-runtime`,
# which is the newest CUDA line the inference host's driver 550 supports, and the only
# combination verified end-to-end on its Pascal GPU (sm_6.1):
#   * 1.19.2 with CUDA 12.4 / cuDNN 9 executes the full YOLOv8n graph (p50 4.45 ms);
#   * 1.23.x runs on CUDA 12.8 and failed the first Conv with cuDNN 209
#     "no kernel image is available for execution on the device";
#   * the pip `nvidia-*-cu12` wheels are NOT used. Pinning them to 12.* resolved to the
#     12.8 build and reproduced the same Conv failure with 1.19.2 on a slim base.
onnxruntime-gpu==1.19.2
